# =============================================================================
# MeshAgent debug v2 二进制解码器 (Windows PowerShell 版)
#
# 用途: 在 Windows 上把 deploy.txt (base64) 还原成 ELF 二进制
#
# 用法:
#   1. 把 meshagent-x86-64-debug-v2.deploy.txt 和这个 .ps1 放在同一目录
#   2. 右键此 .ps1 -> "用 PowerShell 运行"
#   或在 PowerShell 里:
#      cd 到文件所在目录
#      .\decode-deploy.ps1
#
# 输出: 同目录下生成 meshagent_x86-64 (ELF 二进制)
# =============================================================================

$ErrorActionPreference = "Stop"

# 同目录下的 deploy 文件
$deployFile = Join-Path $PSScriptRoot "meshagent-x86-64-debug-v2.deploy.txt"
$outputFile = Join-Path $PSScriptRoot "meshagent_x86-64"
$marker = "9kZ7wQpLv2_BINSTART_xT3"

Write-Host "========================================" -ForegroundColor Cyan
Write-Host "MeshAgent debug v2 解码器" -ForegroundColor Cyan
Write-Host "========================================" -ForegroundColor Cyan
Write-Host ""

# 检查 deploy 文件
if (-not (Test-Path $deployFile)) {
    Write-Host "[错误] 找不到 deploy 文件: $deployFile" -ForegroundColor Red
    Write-Host "请把 meshagent-x86-64-debug-v2.deploy.txt 和此脚本放在同一目录" -ForegroundColor Yellow
    Read-Host "按回车退出"
    exit 1
}

Write-Host "[1/4] 读取 deploy 文件..." -ForegroundColor Green
$lines = Get-Content $deployFile -Encoding UTF8
Write-Host "  文件: $deployFile"
Write-Host "  总行数: $($lines.Count)"

# 找到 marker 行 (从 marker 之后开始是 base64 数据)
$markerIndex = -1
for ($i = 0; $i -lt $lines.Count; $i++) {
    if ($lines[$i] -match "^$marker") {
        $markerIndex = $i
        break
    }
}

if ($markerIndex -eq -1) {
    Write-Host "[错误] 在文件中找不到标记行: $marker" -ForegroundColor Red
    Write-Host "文件可能损坏或传输不完整" -ForegroundColor Yellow
    Read-Host "按回车退出"
    exit 1
}

Write-Host "  找到标记行 (第 $($markerIndex + 1) 行)"
Write-Host "  base64 数据行数: $($lines.Count - $markerIndex - 1)"

# 提取 base64 数据 (marker 行之后的所有行)
$b64Lines = $lines[($markerIndex + 1)..($lines.Count - 1)]
# 拼接成单一字符串, 去掉所有空白/换行/回车
$b64String = ($b64Lines -join "").Replace("`r", "").Replace("`n", "").Replace(" ", "").Replace("`t", "")

Write-Host ""
Write-Host "[2/4] 解码 base64..." -ForegroundColor Green
try {
    $bytes = [System.Convert]::FromBase64String($b64String)
} catch {
    Write-Host "[错误] base64 解码失败: $_" -ForegroundColor Red
    Write-Host "文件内容可能损坏" -ForegroundColor Yellow
    Read-Host "按回车退出"
    exit 1
}

Write-Host "  解码成功, 字节数: $($bytes.Length)"

Write-Host ""
Write-Host "[3/4] 校验..." -ForegroundColor Green

# 检查 ELF 魔数 (前4字节: 7f 45 4c 46)
if ($bytes.Length -ge 4 -and $bytes[0] -eq 0x7f -and $bytes[1] -eq 0x45 -and $bytes[2] -eq 0x4c -and $bytes[3] -eq 0x46) {
    Write-Host "  ELF 魔数: OK (7f 45 4c 46)" -ForegroundColor Green
} else {
    Write-Host "[警告] ELF 魔数不正确!" -ForegroundColor Red
    Write-Host "  前4字节: $([string]::Format('{0:X2} {1:X2} {2:X2} {3:X2}', $bytes[0], $bytes[1], $bytes[2], $bytes[3]))" -ForegroundColor Yellow
    Write-Host "  期望:     7F 45 4C 46" -ForegroundColor Yellow
}

# 检查大小
$expectedSize = 3773840
if ($bytes.Length -eq $expectedSize) {
    Write-Host "  大小: OK ($($bytes.Length) 字节)" -ForegroundColor Green
} else {
    Write-Host "[警告] 大小不符!" -ForegroundColor Yellow
    Write-Host "  实际: $($bytes.Length), 期望: $expectedSize" -ForegroundColor Yellow
}

# 计算 MD5
$md5 = [System.Security.Cryptography.MD5]::Create()
$hashBytes = $md5.ComputeHash($bytes)
$md5Str = [System.BitConverter]::ToString($hashBytes).Replace("-", "").ToLower()
$expectedMd5 = "8d43e4768711beccb696d8fd7a8ad631"

Write-Host "  MD5: $md5Str" -ForegroundColor $(if ($md5Str -eq $expectedMd5) {"Green"} else {"Yellow"})
Write-Host "  期望: $expectedMd5" -ForegroundColor Gray

if ($md5Str -eq $expectedMd5) {
    Write-Host "  MD5 校验: 通过!" -ForegroundColor Green
} else {
    Write-Host "  [警告] MD5 不匹配 (可能仍可用, 但建议重新传输)" -ForegroundColor Yellow
}

Write-Host ""
Write-Host "[4/4] 写入二进制..." -ForegroundColor Green
[System.IO.File]::WriteAllBytes($outputFile, $bytes)
Write-Host "  已写入: $outputFile" -ForegroundColor Green

Write-Host ""
Write-Host "========================================" -ForegroundColor Cyan
if ($md5Str -eq $expectedMd5) {
    Write-Host "  解码成功! 二进制完整无损" -ForegroundColor Green
} else {
    Write-Host "  解码完成, 但 MD5 不符 (见上方警告)" -ForegroundColor Yellow
}
Write-Host "========================================" -ForegroundColor Cyan
Write-Host ""
Write-Host "下一步: 把 meshagent_x86-64 复制到 meshcentral 的 agents 目录" -ForegroundColor White
Write-Host "  通常是: mesh-central-nh\agents\meshagent_x86-64" -ForegroundColor Gray
Write-Host "  然后重新生成 hash:" -ForegroundColor White
Write-Host "  cd mesh-central-nh\agents && node hashagents.js" -ForegroundColor Gray
Write-Host "  重启 meshcentral 服务" -ForegroundColor White
Write-Host ""
Write-Host "UOS agent 会在下次检查更新时自动拉取新二进制。" -ForegroundColor Gray
Write-Host "或在 web 端手动触发该设备的 agent 更新。" -ForegroundColor Gray
Write-Host ""
Read-Host "按回车退出"
